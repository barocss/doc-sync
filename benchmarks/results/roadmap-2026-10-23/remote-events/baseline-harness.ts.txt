import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
import {readFileSync, writeFileSync} from 'node:fs';
import {performance} from 'node:perf_hooks';
import {Doc} from '../../../packages/doc-sync/src/crdt/doc.ts';
import {SyncClient} from '../../../packages/doc-sync/src/sync/client.ts';
import {applyEdits} from '../../../packages/doc-sync/src/crdt/model.ts';
import type {Schema} from '../../../packages/doc-sync/src/crdt/schema.ts';
import type {EditOp} from '../../../packages/doc-sync/src/crdt/model.ts';
import type {ServerEvent} from '../../../packages/doc-sync/src/sync/protocol.ts';
const schema: Schema = {root:'content', marks:['bold'], nodes:{content:{shape:'flow',children:['paragraph'],default:'paragraph'},paragraph:{shape:'text',children:[]}}};
const rows=[];
for (const chars of [10000,100000]) {
 const count=Math.ceil(chars/160);const seed=new Doc('seed',schema);
 seed.edit([{op:'delete',at:[0],count:1},{op:'insert',at:[0],content:Array.from({length:count},(_,i)=>({type:'paragraph',runs:[{text:'x'.repeat(Math.min(160,chars-i*160)),marks:{}}]}))}]);
 let receive:(e:ServerEvent)=>void=()=>{};
 const client=new SyncClient('receiver',{subscribe:(_s,cb)=>{receive=cb;return()=>{};},send:async()=>{},sendPresence:()=>{}},{schema});
 client.connect();receive({type:'hello',serverSeq:1,snapshot:seed.snapshot(),delSeqs:[],presence:[]});client.disconnect();
 const sender=Doc.fromSnapshot('sender',seed.snapshot(),schema);
 let events:EditOp[]=[];client.onModelEvents=e=>{events=e;};let mirror=client.doc.toJSON();
 const read=client.doc.toJSON.bind(client.doc);let reads=0;client.doc.toJSON=()=>{reads++;return read();};
 const times:number[]=[];const readCounts:number[]=[];
 for(let i=0;i<35;i++) {
  const ops=sender.edit([{op:'insert',at:[Math.floor(count/2),50+i],content:'a'}]).ops;
  events=[];reads=0;const start=performance.now();receive({type:'change',envelope:{serverSeq:i+2,change:{actor:'sender',seq:i+1,ops}}});const elapsed=performance.now()-start;
  if(i>=5){times.push(elapsed);readCounts.push(reads);}
  mirror=applyEdits(schema,mirror,events);assert.deepEqual(mirror,read());
 }
 assert.deepEqual(mirror,sender.toJSON());assert.equal(client.eventStats.fallbacks,0);
 const sorted=[...times].sort((a,b)=>a-b);
 rows.push({chars,paragraphs:count,samples:30,warmups:5,p50:sorted[15],p95:sorted[28],max:sorted.at(-1),fullModelReads:readCounts,correct:true,fallbacks:client.eventStats.fallbacks,rawMs:times});
}
const files=['packages/doc-sync/src/sync/client.ts','packages/doc-sync/src/crdt/doc.ts','packages/doc-sync/src/crdt/model.ts'];
const result={at:new Date().toISOString(),node:process.version,metric:'synchronous server-event receipt through model-events callback; excludes editor mirror application, network, DOM',unit:'ms',rows,sourceHashes:files.map(path=>({path,sha256:createHash('sha256').update(readFileSync(path)).digest('hex')}))};
writeFileSync(process.argv[2],JSON.stringify(result,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify(rows.map(({rawMs,fullModelReads,...row})=>({...row,fullModelReads:[...new Set(fullModelReads)]})),null,2));
